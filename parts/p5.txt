function prepPlannedAmountUntil(target,id){
  let total=0;
  const pending=(data.prepPlan||[]).filter(x=>!x.done).slice().sort((a,b)=>String(a.date||'').localeCompare(String(b.date||'')) || (data.prepPlan.indexOf(a)-data.prepPlan.indexOf(b)));
  for(const p of pending){
    for(const r of p.ingredients||[])if(r.id===id){const src=data.stock.find(x=>x.id===id);if(src&&!isQtyUnknown(src))total+=Number(r.amount||0)}
    if(p.id===target.id)break;
  }
  return total;
}
function prepIngredientText(p){
  if(!p.ingredients?.length)return '<span class="small">원료 연결 없음</span>';
  return p.ingredients.map(r=>{
    const src=data.stock.find(x=>x.id===r.id),name=src?.name||r.name||'삭제된 재고',unit=src?.unit||r.unit||'';
    if(src&&isQtyUnknown(src))return `<div>• ${esc(name)} <span class="small">(수량 모름 · 차감 안 함)</span></div>`;
    if(!src)return `<div style="color:#b45309">• ${esc(name)} <span class="small">(현재 재고 없음)</span></div>`;
    const now=Number(src.qty||0),amount=Number(r.amount||0);
    if(p.done)return `<div>• ${esc(name)} ${amount}${unit?' '+esc(unit):''} <span class="small">(현재 ${now}${unit?' '+esc(unit):''})</span></div>`;
    const projected=now-prepPlannedAmountUntil(p,r.id);
    return projected>=0?`<div>• ${esc(name)} ${amount}${unit?' '+esc(unit):''} <span class="small">(현재 ${now}${unit?' '+esc(unit):''} → ${projected}${unit?' '+esc(unit):''} 남음)</span></div>`:`<div style="color:#b45309">• ${esc(name)} ${amount}${unit?' '+esc(unit):''} <span class="small">(현재 ${now}${unit?' '+esc(unit):''} → ${Math.abs(projected)}${unit?' '+esc(unit):''} 부족)</span></div>`;
  }).join('');
}
function prepRecipeIngredientText(r){

}

function finishPrep(id){

  const shortages=[];
  for(const r of p.ingredients||[]){const s=data.stock.find(x=>x.id===r.id);if(!s){shortages.push(r.name||'원료');continue}if(isQtyUnknown(s))continue;if(Number(s.qty||0)<Number(r.amount||0))shortages.push(s.name||r.name||'원료')}
  const msg=shortages.length?`재고가 부족한 원료가 있어요: ${shortages.join(', ')}\n그래도 가능한 수량만 차감하고 밀프랩을 추가할까요?`:'원료 재고를 차감하고 완성된 밀프랩을 재고에 추가할까요?';
  if(!confirm(msg))return;
  p.deducted=[];
  for(const r of p.ingredients||[]){const s=data.stock.find(x=>x.id===r.id);if(!s||isQtyUnknown(s))continue;const before=Number(s.qty||0),used=Math.min(before,Number(r.amount||0));s.qty=Math.max(0,before-used);p.deducted.push({id:r.id,amount:used})}
  const mealId=uid();data.meal.push({id:mealId,name:p.name,qty:Number(p.qty||0),qtyUnknown:false,unit:p.unit||'끼',location:p.location||'냉동',category:'',date:'',madeDate:p.date||todayISO(),note:p.note||''});p.createdMealId=mealId;p.done=true;save();
}

function openMealGroupLoad(id=null){
  const arr=(data.mealGroup||[]).slice().sort((a,b)=>String(a.name||'').localeCompare(String(b.name||''),'ko'));
  if(!arr.length){alert('먼저 “식단 만들기”에서 식단을 저장해주세요.');return}
  $('mealGroupSelect').innerHTML=arr.map(g=>`<option value="${esc(g.id)}" ${id===g.id?'selected':''}>${esc(g.name)}</option>`).join('');
  if(id)$('mealGroupSelect').value=id;
  $('mealGroupTargetDate').value=selectedDate||todayISO();
  refreshMealGroupLoadPreview();
  $('mealGroupLoadDialog').showModal();
}

});

  $('mealGroupLoadDialog').close();save();
});

function render(){
  $('mealCount').textContent=data.meal.filter(x=>Number(x.qty)>0).length;
  $('planCount').textContent=data.plan.filter(x=>!x.done&&x.date>=todayISO()).length;
  $('prepPlanCount').textContent=(data.prepPlan||[]).filter(x=>!x.done).length;

  const isPlan=currentSection==='meal'&&currentMealTab==='plan';
  const isMealGroups=currentSection==='meal'&&currentMealTab==='groups';
  const isFridge=currentSection==='fridge';
  const isPrep=currentSection==='prep';

  $('fridgeTabs').classList.toggle('hide',!isFridge);
  $('mealTabs').classList.toggle('hide',currentSection!=='meal');
  document.querySelectorAll('#mealTabs .sub-tab').forEach(b=>b.classList.toggle('active',b.dataset.mealTab===currentMealTab));
  const fridgeStockView=isFridge&&currentTab==='stock';
  $('fridgeHeading').classList.toggle('hide',!fridgeStockView);

  $('inventoryToolbar').classList.toggle('hide',isPlan||isMealGroups||isPrep);
  $('inventoryToolbar').classList.toggle('fridge-mode',fridgeStockView);
  $('calendarSection').classList.toggle('hide',!isPlan);
  $('mealGroupSection').classList.toggle('hide',!isMealGroups);
  $('prepSection').classList.toggle('hide',!isPrep);
  if(isPrep){
    $('prepRecipesPane').classList.toggle('hide',currentPrepTab!=='recipes');
    $('prepPlansPane').classList.toggle('hide',currentPrepTab!=='plans');
    document.querySelectorAll('#prepTabs .sub-tab').forEach(b=>b.classList.toggle('active',b.dataset.prepTab===currentPrepTab));
  }
  list.classList.toggle('hide',isPlan||isMealGroups||isPrep);

  renderCategoryTabs();
  $('fridgeHeadingTitle').textContent=stockLocation;
  const compartmentCount=data.stock.filter(x=>x.location===stockLocation&&(isQtyUnknown(x)||Number(x.qty)>0)).length;
  $('fridgeHeadingCount').textContent=`${compartmentCount}개 품목`;

  $('addBtn').style.display='block';
  $('addBtn').textContent=isPrep?(currentPrepTab==='recipes'?'＋ 밀프랩 저장':'＋ 계획 추가'):(isMealGroups?'＋ 식단 만들기':(isPlan?'＋ 식단 추가':'＋ 추가'));

  if(isPrep)renderPrep();else if(isMealGroups)renderMealGroups();else if(isPlan)renderCalendar();else renderInventory();
}

function changeQty(type,id,delta){
  const x=data[type].find(v=>v.id===id);if(!x)return;
  if(type==='stock'&&isQtyUnknown(x))return;
  x.qty=Math.max(0,Number(x.qty||0)+delta);save();
}

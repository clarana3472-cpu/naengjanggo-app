
    if(!unknown&&amount<=0)return null;
    return {type,id,amount,name:src?.name||'',unit:src?.unit||'',noDeduct:unknown};
  }).filter(Boolean);
  const old=editingPlanId?data.plan.find(x=>x.id===editingPlanId):null;
  const obj={id:editingPlanId||uid(),date:$('planDate').value,slot:$('planSlot').value,note:$('planNote').value.trim(),ingredients,done:old?.done||false,deducted:old?.deducted||[]};
  if(!obj.date)return;
  if(editingPlanId){const i=data.plan.findIndex(x=>x.id===editingPlanId);data.plan[i]=obj}else data.plan.push(obj);
  selectedDate=obj.date;calendarAnchor=parseLocalDate(selectedDate);
  $('planDialog').close();save();
});

function eatPlan(id){
  const p=data.plan.find(x=>x.id===id);if(!p||p.done)return;
  const shortages=[];
  for(const r of p.ingredients||[]){
    const s=findSource(r);
    if(!s){shortages.push(r.name||'재고');continue}
    if(r.type==='stock'&&isQtyUnknown(s))continue;
    if(Number(s.qty||0)<Number(r.amount||0))shortages.push(s.name||r.name||'재고');
  }
  const hasUnknown=(p.ingredients||[]).some(r=>{const s=findSource(r);return r.type==='stock'&&isQtyUnknown(s)});
  const normalMsg=hasUnknown
    ? '먹은 것으로 표시할까요? “수량 모름” 재고는 그대로 두고, 수량이 있는 재고만 차감됩니다.'
    : '먹은 것으로 표시하고 연결한 재고를 차감할까요?';
  const msg=shortages.length?`재고가 부족한 항목이 있어요: ${shortages.join(', ')}\n그래도 가능한 수량만 차감하고 완료할까요?`:normalMsg;
  if(!confirm(msg))return;
  p.deducted=[];
  for(const r of p.ingredients||[]){
    const src=findSource(r);if(!src)continue;
    if(r.type==='stock'&&isQtyUnknown(src))continue;
    const before=Number(src.qty||0),used=Math.min(before,Number(r.amount||0));
    src.qty=Math.max(0,before-used);p.deducted.push({type:r.type,id:r.id,amount:used});
  }
  p.done=true;save();
}
function undoPlan(id){
  const p=data.plan.find(x=>x.id===id);if(!p||!p.done)return;
  if(!confirm('완료를 취소하고 차감했던 재고를 다시 돌려놓을까요?'))return;
  for(const r of p.deducted||[]){const src=findSource(r);if(src&&!isQtyUnknown(src))src.qty=Number(src.qty||0)+Number(r.amount||0)}
  p.done=false;p.deducted=[];save();
}
function removePlan(id){if(confirm('이 식단을 삭제할까요?')){data.plan=data.plan.filter(x=>x.id!==id);save()}}
window.eatPlan=eatPlan;window.undoPlan=undoPlan;window.removePlan=removePlan;
